public static class Consumer
{
    public static bool Check(bool first, bool second, bool third)
    {
        bool CheckCondition()
        {
            if (!(first))
            {
                return false;
            }

            if (!(second))
            {
                return false;
            }

            if (!(third))
            {
                return false;
            }

            return true;
        }

        return CheckCondition();
    }
}
