public static class Consumer
{
    private static int Use(int value) => value;
    public static int Compute(bool flag)
    {
        int CalculateValue()
        {
            int value;
            if (flag)
            {
                value = 1;
            }
            else
            {
                value = 2;
            }

            return value;
        }

        return Use(CalculateValue());
    }
}
